#Crit mult


crit_mult_average = 1 + min(final_stats[crit_rate],1.0) * final_stats[crit_dmg]

crit_mult_absolute = 1 + final_stats[crit_dmg]




#Def mult

enemy_def = 500 + enemy_stats[enemy_lv] * 5

def_dmg_mitigation = enemy_def / (enemy_def + 500 + "event owner's level" * 500)

def_mult = 1 - def_dmg_mitigation


#res mult

net_res =  enemy_stats[(element of the hit)_res] - final_stats.[(element of the hit)_shred]


if net_res < 0, res_mult = 1 - net_res / 2
if 0 < net_res < 0.75, res_mult = 1 - net_res
if net_res > 0.75, res_mult = 1/(1 + 4 * net_res)

# final standard damage calcs

non_crit = (Base damage + Quill) * bonus_mult * enemy_def * res_mult

crit_damage = (Base damage + Quill) * bonus_mult * crit_mult_absolute * enemy_def * res_mult

average_damage = (Base damage + Quill) * bonus_mult * crit_mult_average * enemy_def * res_mult

effective_average_damage = (Base damage + Quill) * bonus_mult * crit_mult_average * enemy_def
